CAMPOS_TEXTO: Dict[str, str] = {
    "nombre": "nombre",
    "apellido": "apellido",
}


CAMPOS_NUMERICOS: Dict[str, str] = {
    "dni": "dni",
}


SELECT_SOCIOS = (
    "id_persona, nombre, apellido, dni, telefono, "
    "socio!inner("
    "id_socio, codigo_credencial, fecha_alta, "
    "cuota(id_cuota, fecha_vencimiento, estado, monto)"
    ")"
)



def _terminos_de_busqueda(buscar: Optional[str]) -> List[str]:
    """Limpia caracteres especiales y devuelve hasta 5 palabras."""
    if not buscar or not buscar.strip():
        return []
    limpio = re.sub(r'[,()%*_"\\]', " ", buscar.strip())
    return limpio.split()[:5]

def _condiciones_para(palabra: str) -> List[str]:
    condiciones = [f"{col}.ilike.%{palabra}%" for col in CAMPOS_TEXTO.values()]
    if palabra.isascii() and palabra.isdigit() and len(palabra) <= 9:
        condiciones += [f"{col}.eq.{palabra}" for col in CAMPOS_NUMERICOS.values()]
    return condiciones

def _ultima_cuota(cuotas: Any) -> Optional[Dict[str, Any]]:
    if not cuotas:
        return None
    if isinstance(cuotas, dict):
        return cuotas
    return max(cuotas, key=lambda c: c.get("fecha_vencimiento") or "")

def _texto(valor: Any) -> Optional[str]:
    """Convierte a str (o None) para que Pydantic no rechace nÃºmeros."""
    return None if valor is None else str(valor)

def _armar_socio(persona: Dict[str, Any]) -> Dict[str, Any]:
    """Aplana persona + socio + Ãºltima cuota en un solo diccionario."""
    socio = persona.get("socio") or {}
    if isinstance(socio, list):
        socio = socio[0] if socio else {}

    cuota = _ultima_cuota(socio.get("cuota")) or {}

    return {
        "id_socio": socio.get("id_socio"),
        "codigo_credencial": _texto(socio.get("codigo_credencial")),
        "nombre": persona.get("nombre"),
        "apellido": persona.get("apellido"),
        "dni": _texto(persona.get("dni")),
        "telefono": _texto(persona.get("telefono")),
        "fecha_alta": socio.get("fecha_alta"),
        "fecha_vencimiento": cuota.get("fecha_vencimiento"),
        "estado_cuota": cuota.get("estado") or "sin_cuota",
        "monto_cuota": cuota.get("monto"),
    }


def _error_bd(e: APIError) -> HTTPException:
    """Loguea el detalle real y devuelve un error genÃ©rico al cliente."""
    logger.error("Supabase APIError | code=%s | message=%s | details=%s",
                 e.code, e.message, e.details)
    return HTTPException(
        status_code=status.HTTP_502_BAD_GATEWAY,
        detail="Error al consultar los socios",
    )




@router.get("", response_model=ListadoSociosResponse)
def obtener_socios_paginados(
    page: int = Query(1, ge=1, description="NÃºmero de pÃ¡gina"),
    limit: int = Query(10, ge=1, le=100, description="Cantidad de socios por pÃ¡gina"),
    buscar: Optional[str] = Query(
        None, max_length=50, description="Texto a buscar en nombre, apellido o DNI"
    ),
    supabase: Client = Depends(get_supabase),
):
    inicio = (page - 1) * limit
    fin = inicio + limit - 1

    consulta = supabase.table("persona").select(SELECT_SOCIOS, count="exact")

    
    for palabra in _terminos_de_busqueda(buscar):
        consulta = consulta.or_(",".join(_condiciones_para(palabra)))

    try:
        respuesta = (
            consulta
            .order("apellido", desc=False)
            .order("id_persona", desc=False)
            .range(inicio, fin)
            .execute()
        )
        filas = respuesta.data or []
        total_socios = respuesta.count or 0

    except APIError as e:
        if e.code != "PGRST103":
            raise _error_bd(e)
        
        try:
            respuesta = consulta.range(0, 0).execute()
        except APIError as e2:
            raise _error_bd(e2)
        filas = []
        total_socios = respuesta.count or 0

    except Exception:
        logger.exception("Error no controlado al obtener los socios")
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="Error interno al obtener los socios",
        )

    datos = [_armar_socio(fila) for fila in filas]
    total_paginas = math.ceil(total_socios / limit) if total_socios else 0

    return {
        "datos": datos,
        "paginacion": {
            "paginaActual": page,
            "elementosPorPagina": limit,
            "totalPaginas": total_paginas,
            "totalSocios": total_socios,
        },
    }

def _generar_hash_y_salt(password: str) -> tuple[str, str]:
    """
    Genera un salt aleatorio de 16 bytes y el hash PBKDF2-HMAC-SHA256.
    Devuelve (hash_hex, salt_bytea_hex) listo para insertar en PostgreSQL/Supabase.
    """
    salt_bytes = os.urandom(16)
    hash_hex = hashlib.pbkdf2_hmac(
        "sha256",
        password.encode("utf-8"),
        salt_bytes,
        100_000,
    ).hex()
    
    salt_bytea = f"\\x{salt_bytes.hex()}"
    return hash_hex, salt_bytea


class UsuarioBase(BaseModel):
    """Esquema base con los atributos comunes a todos los usuarios."""
    model_config = ConfigDict(str_strip_whitespace=True)

    nombre: str = Field(min_length=1, description="Nombre de la persona")
    apellido: str = Field(min_length=1, description="Apellido de la persona")
    dni: str = Field(pattern=r"^\d{7,9}$", description="NÃºmero de documento")
    telefono: Optional[str] = Field(
        default=None,
        description="TelÃ©fono (se formatea automÃ¡ticamente a XX-XXXX-XXXX, ej: 11-5555-1000)",
    )
    fecha_nacimiento: Optional[date] = Field(
        default=None,
        description="Fecha de nacimiento en formato YYYY-MM-DD (ej: 1998-05-20)",
    )
    email: EmailStr
    password: str = Field(min_length=6, description="ContraseÃ±a de al menos 6 caracteres")
    rol: str = Field(description="Rol asignado al usuario")
    id_sede: Optional[int] = Field(
        default=None,
        description="ID de la sede (obligatorio para socio, opcional para otros)",
    )

    @field_validator("telefono", mode="before")
    @classmethod
    def formatear_telefono(cls, v: Optional[str]) -> Optional[str]:
        if v is None:
            return None
        v_str = str(v).strip()
        if not v_str:
            return None

        
        digitos = re.sub(r"\D", "", v_str)

        
        if digitos.startswith("549") and len(digitos) == 13:
            digitos = digitos[3:]
        elif digitos.startswith("54") and len(digitos) == 12:
            digitos = digitos[2:]
        elif digitos.startswith("0") and len(digitos) == 11:
            digitos = digitos[1:]

        
        if len(digitos) == 8:
            digitos = f"11{digitos}"

        if len(digitos) != 10:
            raise ValueError(
                "El telÃ©fono debe contener 10 dÃ­gitos (ej: 1155551000 o 11-5555-1000)."
            )

        # Retornar en formato XX-XXXX-XXXX (ej: 11-7537-4712)
        return f"{digitos[:2]}-{digitos[2:6]}-{digitos[6:]}"

    @field_validator("id_sede", mode="before")
    @classmethod
    def normalizar_sede(cls, v):
        if v is None or v == "" or (isinstance(v, int) and v <= 0):
            return None
        return v

    @field_validator("email")
    @classmethod
    def email_minusculas(cls, v: EmailStr) -> str:
        return v.lower()


def _rollback(supabase: Client, user_id: Optional[str], id_persona: Optional[int], rol: str):
    """Deshace los registros creados en orden inverso si ocurre algÃºn fallo."""
    try:
        if id_persona is not None:
            supabase.table(rol).delete().eq("id_persona", id_persona).execute()
            supabase.table("usuario").delete().eq("id_persona", id_persona).execute()
            supabase.table("persona").delete().eq("id_persona", id_persona).execute()
        if user_id:
            supabase.auth.admin.delete_user(user_id)
    except Exception:
        logger.exception("Error durante rollback (id_persona=%s, user_id=%s)", id_persona, user_id)


def _existe_activo(supabase: Client, tabla: str, columna: str, valor) -> bool:
    """Verifica si un registro existe y se encuentra activo en la base de datos."""
    res = supabase.table(tabla).select(columna).eq(columna, valor).eq("activo", True).execute()
    return bool(res.data)


def crear_usuario_completo(supabase: Client, datos: UsuarioBase) -> dict:
    """LÃ³gica unificada de negocio y persistencia en Supabase."""
    rol = datos.rol
    id_deporte = getattr(datos, "id_deporte", None) if rol in ROLES_CON_DEPORTE else None

    
    if rol in ROLES_SEDE_OBLIGATORIA and datos.id_sede is None:
        raise HTTPException(
            status_code=400,
            detail="El socio debe tener una sede vÃ¡lida asignada (id_sede mayor a 0).",
        )
    if rol in ROLES_CON_DEPORTE and id_deporte is None:
        raise HTTPException(
            status_code=400,
            detail="El representante debe tener un deporte asignado (id_deporte mayor a 0).",
        )

    
    grupo = supabase.table("grupo_usuario").select("id_grupo").ilike("nombre", rol).execute()
    if not grupo.data:
        raise HTTPException(
            status_code=400,
            detail=f"No existe el grupo para el rol '{rol}' en la tabla grupo_usuario.",
        )
    id_grupo = grupo.data[0]["id_grupo"]

    
    if id_deporte is not None and not _existe_activo(supabase, "deporte", "id_deporte", id_deporte):
        raise HTTPException(
            status_code=400,
            detail=f"El deporte con ID {id_deporte} no existe o estÃ¡ inactivo.",
        )
    if datos.id_sede is not None and not _existe_activo(supabase, "sede", "id_sede", datos.id_sede):
        raise HTTPException(
            status_code=400,
            detail=f"La sede con ID {datos.id_sede} no existe o estÃ¡ inactiva.",
        )

    
    if supabase.table("persona").select("id_persona").eq("dni", datos.dni).execute().data:
        raise HTTPException(status_code=409, detail="Ya existe una persona registrada con ese DNI.")
    if supabase.table("usuario").select("id_usuario").eq("email", datos.email).execute().data:
        raise HTTPException(status_code=409, detail="Ya existe un usuario registrado con ese email.")

    user_id = None
    id_persona = None
    try:
        
        auth_response = supabase.auth.admin.create_user({
            "email": datos.email,
            "password": datos.password,
            "email_confirm": True,
        })
        user_id = auth_response.user.id

        
        persona_res = supabase.table("persona").insert({
            "nombre": datos.nombre,
            "apellido": datos.apellido,
            "dni": datos.dni,
            "telefono": datos.telefono,
            "fecha_nacimiento": datos.fecha_nacimiento.isoformat() if datos.fecha_nacimiento else None,
        }).execute()
        id_persona = persona_res.data[0]["id_persona"]

        
        hash_hex, salt_bytea = _generar_hash_y_salt(datos.password)

        supabase.table("usuario").insert({
            "id_usuario": user_id,
            "id_persona": id_persona,
            "id_sede": datos.id_sede,
            "id_grupo": id_grupo,
            "email": datos.email,
            "nombre_usuario": f"{datos.nombre} {datos.apellido}",
            "hash": hash_hex,
            "salt": salt_bytea,
        }).execute()

        
        payload = {"id_persona": id_persona}
        if rol == "socio":
            payload["id_sede"] = datos.id_sede
        elif rol == "representante":
            payload["id_deporte"] = id_deporte

        rol_res = supabase.table(rol).insert(payload).execute()
        datos_rol = rol_res.data[0] if rol_res.data else {}

        respuesta = {
            "mensaje": f"{rol.capitalize()} registrado exitosamente en el sistema.",
            "id_usuario": user_id,
            "id_persona": id_persona,
            "rol": rol,
            PK_POR_ROL[rol]: datos_rol.get(PK_POR_ROL[rol]),
        }

        if rol == "socio" and "codigo_credencial" in datos_rol:
            respuesta["codigo_credencial"] = datos_rol["codigo_credencial"]

        return respuesta

    except HTTPException:
        _rollback(supabase, user_id, id_persona, rol)
        raise
    except Exception as e:
        _rollback(supabase, user_id, id_persona, rol)
        logger.exception("Error al registrar el usuario")

        texto = str(e).lower()
        if "duplicate key" in texto or "23505" in texto or "already" in texto:
            raise HTTPException(status_code=409, detail="El DNI o el email ya estÃ¡n registrados.")

        raise HTTPException(
            status_code=500,
            detail="Error al registrar el usuario en la base de datos.",
        )
